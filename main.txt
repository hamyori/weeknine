#include <fstream>
#include <iomanip>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>

struct Producto {
  std::string codigo;
  std::string nombre;
  double precio;
  int existencia;
};

bool validarYConvertir(const std::string &str_precio,
                       const std::string &str_existencia, double &precio,
                       int &existencia) {
  try {
    size_t idx_p, idx_e;
    precio = std::stod(str_precio, &idx_p);
    if (idx_p != str_precio.length() || precio <= 0.0)
      return false;

    existencia = std::stoi(str_existencia, &idx_e);
    if (idx_e != str_existencia.length() || existencia < 0)
      return false;

    return true;
  } catch (...) {
    return false;
  }
}

int main() {
  std::ifstream archivo("datos/productos.csv");
  if (!archivo.is_open()) {
    std::cerr << "Error: No se pudo abrir datos/productos.csv\n";
    return 1;
  }

  std::string linea;
  // Omitir cabecera
  if (!std::getline(archivo, linea)) {
    std::cerr << "Error: Archivo vacío o corrupto.\n";
    return 1;
  }

  std::vector<Producto> productos_validos;
  int validos = 0;
  int invalidos = 0;
  double valor_total = 0.0;

  while (std::getline(archivo, linea)) {
    if (linea.empty())
      continue;

    std::stringstream ss(linea);
    std::string codigo, nombre, str_precio, str_existencia;

    std::getline(ss, codigo, ',');
    std::getline(ss, nombre, ',');
    std::getline(ss, str_precio, ',');
    std::getline(ss, str_existencia, ',');

    double precio = 0.0;
    int existencia = 0;

    if (codigo.empty() || nombre.empty()) {
      invalidos++;
      continue;
    }

    if (validarYConvertir(str_precio, str_existencia, precio, existencia)) {
      validos++;
      valor_total += (precio * existencia);
      productos_validos.push_back({codigo, nombre, precio, existencia});
    } else {
      invalidos++;
    }
  }
  archivo.close();

  // Consulta al usuario
  std::string codigo_busqueda;
  std::cout << "Ingrese código de producto a buscar: ";
  std::cin >> codigo_busqueda;

  bool hallado = false;
  Producto resultado;
  for (const auto &p : productos_validos) {
    if (p.codigo == codigo_busqueda) {
      hallado = true;
      resultado = p;
      break;
    }
  }

  if (hallado) {
    std::cout << "Encontrado: " << resultado.nombre << " | Precio: Q"
              << resultado.precio << " | Existencia: " << resultado.existencia
              << "\n";
  } else {
    std::cout << "El producto con código " << codigo_busqueda
              << " no existe.\n";
  }

  // Volcado de reporte
  std::ofstream reporte("reportes/resumen.txt");
  if (!reporte.is_open()) {
    std::cerr << "Aviso: No se pudo crear reportes/resumen.txt (asegúrate de "
                 "que la carpeta 'reportes' exista).\n";
    return 1;
  }

  reporte << std::fixed << std::setprecision(2);
  reporte << "Total de registros válidos: " << validos << "\n";
  reporte << "Total de registros inválidos: " << invalidos << "\n";
  reporte << "Valor total del inventario: Q" << valor_total << "\n";
  if (hallado) {
    reporte << "Búsqueda (" << codigo_busqueda << "): " << resultado.nombre
            << " | Precio: Q" << resultado.precio
            << " | Existencia: " << resultado.existencia << "\n";
  } else {
    reporte << "Búsqueda (" << codigo_busqueda
            << "): Producto no encontrado.\n";
  }
  reporte.close();

  return 0;
}
